cmake_minimum_required(VERSION 3.8)
project(carkit_vehicle_runtime)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(ackermann_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
add_executable(imu_rate_limit src/imu_rate_limit.cpp)
ament_target_dependencies(imu_rate_limit rclcpp sensor_msgs)
install(TARGETS imu_rate_limit DESTINATION lib/${PROJECT_NAME})
option(CARKIT_ENABLE_VESC_TELEMETRY "Build optional VESC adapters" ON)
if(CARKIT_ENABLE_VESC_TELEMETRY)
  find_package(vesc_msgs QUIET)
  if(vesc_msgs_FOUND)
    foreach(target realsense_wheel_odometry vesc_speed_control)
      add_executable(${target} src/${target}.cpp)
      target_include_directories(${target} PRIVATE include)
      ament_target_dependencies(${target} rclcpp sensor_msgs nav_msgs ackermann_msgs tf2 tf2_ros vesc_msgs)
      install(TARGETS ${target} DESTINATION lib/${PROJECT_NAME})
    endforeach()
  else()
    message(STATUS "VESC messages unavailable; building sensor-only runtime")
  endif()
endif()
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_motion_math test/test_motion_math.cpp)
  target_include_directories(test_motion_math PRIVATE include)
endif()
ament_package()
